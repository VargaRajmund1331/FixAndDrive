sw10
vlan 10
name IRODA
int range fa0/1-24
switchport mode Access
switchport Access vlan 10
exit
int range g0/1-2
switchport mode trunk
switchport trunk native vlan 10
exit
int vlan 10
ip add 192.168.10.4 255.255.255.0
no shu
desc IRODA_ADMIN_IP
exit
ip default-gateway 192.168.10.3
